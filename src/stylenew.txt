@import "tailwindcss";

/*
Project color palette, variation B: blue nav.

Given: nav blue #126a97. Kept from before: cream #efecd9, yellow #fbd046,
blue #1c8cc8. Everything else is derived from those, or is a meaning color.
If unsure which to use, use ink or brand.

How the page reads, top to bottom:
  nav (blue, white text)  ->  cream page with white cards  ->  footer (deeper blue)
Yellow is the highlight: badges, nav hover underline, footer links on hover.
*/
@theme {
  /* Navbar background. Text on it is always white. */
  --color-nav: #126a97;

  /* Footer background. A deeper step of the nav blue, so the page opens and
     closes in the same family. Text on it is white, links are white/70,
     link hover and headings accents are yellow (accent). */
  --color-footer: #0d4566;

  /* Page background. White cards sit on top of it. */
  --color-surface: #efecd9;

  /* Action: the only interactive color (buttons, links, focus, active tabs).
     White text on brand is 3.7:1, fine for large or bold text. For small
     text on a brand fill, use brand-dark (it is the nav blue, 5.9:1). */
  --color-brand: #1c8cc8;
  --color-brand-dark: #126a97; /* hover for brand fills, and links on cream/white (5:1) */
  --color-brand-soft: #e3f1f9; /* very light tint for backgrounds and icon circles */

  /* Structure: headings, body text, date tiles, outline buttons.
     A deep navy taken from the blue, since the palette has no dark color. */
  --color-ink: #10324a;

  /* Meaning and highlight only. Never for plain decoration.
     Yellow is the highlight on the blue nav and footer (4:1 and 6.9:1),
     and a badge fill on light pages (text on it is ink). */
  --color-accent: #fbd046; /* badges (Featured, Free), "pending", nav and footer highlights */
  --color-success: #00bfa6; /* paid, available, active */
  --color-danger: #ff3b30; /* errors, sold out */

  /* Darker shades for TEXT on white, cream or tinted backgrounds.
     The bright versions are too light to read at small sizes.
     Pattern: bg-success/10 text-success-dark */
  --color-accent-dark: #7a5a00;
  --color-success-dark: #007f6e;
  --color-danger-dark: #d92d20;
}

/* ---------------------------------------------------------------
   Swiper sliders (Featured, Recommended, Creators)
   Global on purpose: the pagination lives inside the
   swiper-container's shadow DOM, so it can only be styled through
   CSS variables and ::part().
   The prev/next arrows are not styled here: each slider has its own
   buttons in its header, next to "See more".
---------------------------------------------------------------- */
swiper-container {
  --swiper-pagination-color: var(--color-brand);
  --swiper-pagination-bullet-size: 8px;
  --swiper-pagination-bullet-inactive-color: var(--color-ink);
  --swiper-pagination-bullet-inactive-opacity: 0.25;
  --swiper-pagination-bottom: 4px;
}

/* Room under the cards for the dots.
   This has to go on the inner container, not on swiper-container itself:
   the dots are positioned inside it, so padding on the outer element
   leaves them sitting on top of the cards. It also stops the card hover
   shadow from being clipped at the bottom. */
swiper-container::part(container) {
  padding-bottom: 40px;
}

/* Pagination container */
swiper-container::part(pagination) {
  display: flex;
  justify-content: flex-start;
  padding-left: 5px;
}

/* Pagination bullets */
swiper-container::part(bullet) {
  transition: all 0.3s ease;
}

swiper-container::part(bullet-active) {
  width: 22px;
  border-radius: 9999px;
}
